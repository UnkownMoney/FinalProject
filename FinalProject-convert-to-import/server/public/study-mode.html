document.addEventListener('DOMContentLoaded', async () => { let flashcards = []; let currentIndex = 0; // Fetch flashcards from API async function loadFlashcards() { try { const response = await fetch('/api/flashcards'); if (!response.ok) throw new Error('Failed to load cards'); flashcards = await response.json(); updateCardCounter(); showCurrentCard(); } catch (err) { console.error('Error:', err); document.getElementById('study-front').textContent = 'Error loading cards'; } } // Navigation functions function showCurrentCard() { if (flashcards.length === 0) return; document.getElementById('study-front').textContent = flashcards[currentIndex].question; document.getElementById('study-back').textContent = flashcards[currentIndex].answer; document.querySelector('.study-card').classList.remove('flipped'); } function updateCardCounter() { document.getElementById('current-card').textContent = currentIndex + 1; document.getElementById('total-cards').textContent = flashcards.length; } // Event listeners document.getElementById('study-prev').addEventListener('click', () => { if (currentIndex > 0) { currentIndex--; showCurrentCard(); updateCardCounter(); } }); document.getElementById('study-next').addEventListener('click', () => { if (currentIndex < flashcards.length - 1) { currentIndex++; showCurrentCard(); updateCardCounter(); } }); document.getElementById('study-flip').addEventListener('click', () => { document.querySelector('.study-card').classList.toggle('flipped'); }); document.getElementById('back-to-list').addEventListener('click', () => { window.location.href = '/'; }); // Initial load loadFlashcards(); });
